# INSTALAÇÃO NO WINDOWS:
# dlib quase nunca tem wheel pré-compilada no PyPI para Windows,
# então ele tenta compilar do zero (precisa de CMake + Visual C++
# Build Tools). O caminho mais simples é instalar as libs pesadas
# via conda-forge, que já tem binários prontos:
#
#   conda create -n perception python=3.11 -y
#   conda activate perception
#   conda install -c conda-forge dlib numpy opencv -y
#
# IMPORTANTE: se usou o conda acima, NÃO rode
# "pip install -r requirements.txt" direto (ele tentaria reinstalar
# dlib/numpy/opencv via pip e falharia por falta de CMake).
# Em vez disso, instale só o restante:
#
#   pip install imutils PyYAML scipy
#
# Se a janela do OpenCV não abrir (erro de imshow/GTK), o build do
# conda pode ser "headless" — troque só o cv2 por uma versão com GUI:
#
#   pip install opencv-python
#
# No Linux/Mac normalmente `pip install -r requirements.txt` direto
# já funciona sem problema (não precisa de conda).
